Usage (React Hook)

Example.tsx 
import { useToast } from '@webmx/ui';

const Example = () => {
  const { toast } = useToast();

  const showNotification = () => {
    toast({
      title: 'Success!',
      description: 'Profile updated successfully',
      type: 'success',
      duration: 5000,
      position: 'top-right',
    });
  };

  return (
    <button onClick={showNotification}>
      Show Toast
    </button>
  );
};

Props Reference

Prop Type Default Description
message string - The main body text of the notification.
type 'success' | 'error' | 'info' | 'warning' 'info' Defines the semantic theme and icon.
duration number 3000 Time in ms before auto-closing. Set to 0 to disable.
position Enum 'top-right' Screen placement for the toast stack.
